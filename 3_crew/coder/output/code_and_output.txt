```python
# Python program to calculate the first 10,000 terms
# of the series 1 - 1/3 + 1/5 - 1/7 + ...

total = 0.0
for i in range(10000):
    denominator = 2 * i + 1
    term = (1 / denominator) if i % 2 == 0 else -(1 / denominator)
    total += term

final_result = total * 4
print(final_result)
```
